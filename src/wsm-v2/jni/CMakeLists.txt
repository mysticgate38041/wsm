cmake_minimum_required(VERSION 3.22)
project(wsm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(WSM_ENGINE_SOURCES
    engine.cpp il2cpp_resolver.cpp aob_scanner.cpp hybrid_resolver.cpp
    feature_flags.cpp dispatcher.cpp payload_worker.cpp)
set(WSM_UNIT_NAMES runtime patch binding reloc sweep flags resolver dispatcher worker pool bus_event restore)

if(ANDROID)
    if(NOT ANDROID_ABI MATCHES "^(x86_64|arm64-v8a)$")
        message(FATAL_ERROR "WSM supports x86_64 and arm64-v8a only")
    endif()
    # r27c's default legacy toolchain sets CMAKE_SYSTEM_VERSION=1 to suppress
    # CMake's NDK handling; ANDROID_PLATFORM_LEVEL is its effective target API.
    if(DEFINED ANDROID_PLATFORM_LEVEL)
        set(WSM_ANDROID_API "${ANDROID_PLATFORM_LEVEL}")
    elseif(DEFINED ANDROID_PLATFORM)
        string(REGEX REPLACE "^android-" "" WSM_ANDROID_API "${ANDROID_PLATFORM}")
    else()
        set(WSM_ANDROID_API "${CMAKE_SYSTEM_VERSION}")
    endif()
    if(NOT WSM_ANDROID_API STREQUAL "26")
        message(FATAL_ERROR "Configure with -DANDROID_PLATFORM=android-26 (effective API: ${WSM_ANDROID_API})")
    endif()
    if(DEFINED CMAKE_ANDROID_STL_TYPE)
        set(WSM_ANDROID_STL "${CMAKE_ANDROID_STL_TYPE}")
    else()
        set(WSM_ANDROID_STL "${ANDROID_STL}")
    endif()
    if(NOT WSM_ANDROID_STL STREQUAL "c++_static")
        message(FATAL_ERROR "Configure with -DANDROID_STL=c++_static")
    endif()
    if(CMAKE_ANDROID_NDK)
        set(WSM_ANDROID_NDK "${CMAKE_ANDROID_NDK}")
    else()
        set(WSM_ANDROID_NDK "${ANDROID_NDK}")
    endif()
    file(READ "${WSM_ANDROID_NDK}/source.properties" WSM_NDK_PROPERTIES)
    if(NOT WSM_NDK_PROPERTIES MATCHES "Pkg\\.Revision[ \t]*=[ \t]*27\\.2\\.12479018([ \r\n]|$)")
        message(FATAL_ERROR "Pinned NDK r27c (27.2.12479018) required")
    endif()
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/libs/${ANDROID_ABI}")
    add_library(wsm_loader SHARED module.cpp)
    add_library(wsm_engine SHARED ${WSM_ENGINE_SOURCES})
    set(WSM_NATIVE_TARGETS wsm_loader wsm_engine)
    target_link_libraries(wsm_loader PRIVATE log dl)
    target_link_libraries(wsm_engine PRIVATE log dl)
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        add_library(wsm_arm64 SHARED ../payload/h64.cpp trampoline_pool.cpp)
        target_link_libraries(wsm_arm64 PRIVATE log dl)
        target_compile_options(wsm_arm64 PRIVATE -Wno-unused-variable)
        list(APPEND WSM_NATIVE_TARGETS wsm_arm64)
    endif()
    foreach(target IN LISTS WSM_NATIVE_TARGETS)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter
            -fno-exceptions -fno-rtti -fvisibility=hidden -fvisibility-inlines-hidden
            -ffunction-sections -fdata-sections)
        # Source entrypoints retain explicit default visibility; archive symbols do not.
        target_link_options(${target} PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL
            -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    endforeach()
else()
    # Portable fixtures exercise generic native contracts, not Android or Unity runtime.
    option(WSM_BUILD_HOST_TESTS "Build and register POSIX host fixtures" ON)
    if(WSM_BUILD_HOST_TESTS)
        if(WIN32)
            message(FATAL_ERROR "Host fixtures require POSIX APIs; use Linux/WSL or an Android cross-build")
        endif()
        find_package(Threads REQUIRED)
        enable_testing()
        foreach(unit IN LISTS WSM_UNIT_NAMES)
            add_executable(wsm_${unit}_test ../tests/${unit}_test.cpp)
            target_compile_options(wsm_${unit}_test PRIVATE -O2 -UNDEBUG -Wall -Wextra -Werror
                -fno-exceptions -fno-rtti)
            target_compile_definitions(wsm_${unit}_test PRIVATE WSM_TEST_TMPDIR="/tmp")
            target_link_libraries(wsm_${unit}_test PRIVATE Threads::Threads)
            if(unit STREQUAL "binding")
                target_sources(wsm_${unit}_test PRIVATE il2cpp_resolver.cpp)
                target_link_libraries(wsm_${unit}_test PRIVATE ${CMAKE_DL_LIBS})
            elseif(unit STREQUAL "flags")
                target_sources(wsm_${unit}_test PRIVATE feature_flags.cpp)
            elseif(unit STREQUAL "resolver")
                target_sources(wsm_${unit}_test PRIVATE il2cpp_resolver.cpp aob_scanner.cpp hybrid_resolver.cpp)
                target_link_libraries(wsm_${unit}_test PRIVATE ${CMAKE_DL_LIBS})
            elseif(unit STREQUAL "dispatcher")
                target_sources(wsm_${unit}_test PRIVATE dispatcher.cpp)
            elseif(unit STREQUAL "worker")
                target_sources(wsm_${unit}_test PRIVATE payload_worker.cpp)
            elseif(unit STREQUAL "pool")
                target_sources(wsm_${unit}_test PRIVATE trampoline_pool.cpp)
            endif()
            add_test(NAME ${unit} COMMAND wsm_${unit}_test)
        endforeach()
    endif()
endif()
